#include <Wire.h> 
#include <LiquidCrystal_I2C.h>
#include <DHT.h>

// 1. Initialize LCD (I2C address 0x27)
LiquidCrystal_I2C lcd(0x27, 16, 2); 

// 2. Initialize DHT22 Sensor
#define DHTPIN 2
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);

// 3. Pin Definitions
const int MQ135_PIN = A0;
const int MQ3_PIN   = A1;
const int MQ7_PIN   = A2;
const int MQ2_PIN   = A3;

const int FAN_PIN    = 9;   // Transistor Signal Pin
const int BUZZER_PIN = 8;   // Alarm Signal Pin
const int LIMIT      = 450; // Gas Threshold Value

// 4. Timer & State Tracking
unsigned long alarmStartTime = 0;
bool gasPresent = false;
bool buzzerActive = false;

void setup() {
  Serial.begin(9600); 
  Wire.begin();
  lcd.init();
  lcd.backlight();
  dht.begin();
  
  pinMode(FAN_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  
  // Startup Splash Screen
  lcd.setCursor(0, 0);
  lcd.print("VIT E-NOSE v4.1");
  lcd.setCursor(0, 1);
  lcd.print("INITIALIZING...");
  Serial.println("--- System Online: Electronic Nose Prototype ---");
  delay(2000);
  lcd.clear();
}

void loop() {
  // Read Analog Gas Sensors
  int s135 = analogRead(MQ135_PIN);
  int s3   = analogRead(MQ3_PIN);
  int s7   = analogRead(MQ7_PIN);
  int s2   = analogRead(MQ2_PIN);

  // Read Environmental Conditions
  float h = dht.readHumidity();
  float t = dht.readTemperature();

  // Evaluate Danger Condition
  bool danger = (s135 > LIMIT || s3 > LIMIT || s7 > LIMIT || s2 > LIMIT);

  // --- ACTUATION & TIMER LOGIC ---
  if (danger) {
    digitalWrite(FAN_PIN, HIGH); // Exhaust fan remains on to purge gas
    
    if (!gasPresent) {
      gasPresent = true;
      buzzerActive = true;
      alarmStartTime = millis();
      digitalWrite(BUZZER_PIN, HIGH); // Trigger Buzzer Alarm
      Serial.println("!! ALERT: Hazardous Gas Level Exceeded !!");
    }
    
    // Silence buzzer after 6 seconds
    if (buzzerActive && (millis() - alarmStartTime >= 6000)) {
      digitalWrite(BUZZER_PIN, LOW);
      buzzerActive = false;
      Serial.println("Status: Alarm Silenced (6s Timeout) - Fan Active");
    }
  } else {
    // Air is clear: Reset actuators
    digitalWrite(FAN_PIN, LOW);
    digitalWrite(BUZZER_PIN, LOW);
    gasPresent = false;
    buzzerActive = false;
  }

  // --- SERIAL MONITOR OUTPUT ---
  Serial.print("MQ135:"); Serial.print(s135);
  Serial.print(" | MQ3:"); Serial.print(s3);
  Serial.print(" | MQ7:"); Serial.print(s7);
  Serial.print(" | MQ2:"); Serial.print(s2);
  Serial.print(" | Temp:"); Serial.print(t);
  Serial.print("C | Hum:"); Serial.print(h);
  Serial.println("%");

  // --- LCD OUTPUT ---
  lcd.setCursor(0, 0);
  lcd.print("T:"); lcd.print(t, 0); lcd.print("C H:"); lcd.print(h, 0); lcd.print("%   ");
  
  lcd.setCursor(0, 1);
  if (s7 > LIMIT)        lcd.print("!! CO ALERT !!  ");
  else if (s2 > LIMIT)   lcd.print("!! SMOKE/LPG !! ");
  else if (s3 > LIMIT)   lcd.print("!! ALCOHOL !!   ");
  else if (s135 > LIMIT) lcd.print("!! BAD AIR !!   ");
  else                   lcd.print("Status: Normal  ");

  delay(500); 
}
